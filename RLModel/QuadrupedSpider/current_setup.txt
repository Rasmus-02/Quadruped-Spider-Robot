from __future__ import annotations

from typing import TYPE_CHECKING

import torch

from isaaclab.envs.mdp import height_scan
from isaaclab.managers import SceneEntityCfg
from isaaclab.sensors import RayCaster

if TYPE_CHECKING:
    from isaaclab.envs import ManagerBasedRLEnv


def corrupted_height_scan(
    env: ManagerBasedRLEnv, 
    sensor_cfg: SceneEntityCfg, 
    offset: float = 0.5,
    noise_std: float = 0.015, 
    dropout_prob: float = 0.05
) -> torch.Tensor:
    """
    Samples the height scan matching Isaac Lab's mdp.height_scan convention
    (sensor_z - hit_z - offset) and applies synthetic noise and dropouts.
    """
    sensor: RayCaster = env.scene.sensors[sensor_cfg.name]
    
    hit_pos_w = sensor.data.ray_hits_w[..., 2]
    sensor_pos_w = sensor.data.pos_w[:, 2].unsqueeze(1)
    
    # Exact match with isaaclab.envs.mdp.height_scan
    heights = sensor_pos_w - hit_pos_w - offset
    
    # Handle infinite rays if any miss the terrain mesh
    heights = torch.nan_to_num(heights, nan=1.0, posinf=1.0, neginf=-1.0)
    
    # Add Gaussian noise
    if noise_std > 0.0:
        heights = heights + torch.randn_like(heights) * noise_std
    
    # Add dropouts (assign max distance reading 1.0)
    if dropout_prob > 0.0:
        dropout_mask = torch.rand_like(heights) < dropout_prob
        heights[dropout_mask] = 1.0
    
    return torch.clamp(heights, min=-1.0, max=1.0)

import torch
from isaaclab.envs import ManagerBasedRLEnv
from isaaclab.managers import SceneEntityCfg
from isaaclab.sensors import RayCaster


def tof_distance_below_minimum(
    env: ManagerBasedRLEnv, minimum_distance: float, sensor_cfg: SceneEntityCfg
) -> torch.Tensor:
    """Terminate when the ToF sensor reads below threshold or clips into the ground."""
    tof: RayCaster = env.scene[sensor_cfg.name]
    
    # Extract world positions
    hit_pos_w = tof.data.ray_hits_w[:, 0, :]
    sensor_pos_w = tof.data.pos_w
    
    # Calculate physical distance from sensor to the hit point
    distances = torch.norm(sensor_pos_w - hit_pos_w, dim=-1)
    
    # In Isaac Lab, a raycast miss defaults to [0.0, 0.0, 0.0]. 
    # If it misses, it means the sensor clipped under the floor mesh.
    missed = (hit_pos_w[:, 0] == 0.0) & (hit_pos_w[:, 1] == 0.0) & (hit_pos_w[:, 2] == 0.0)
    
    # Terminate if the ground is too close, OR if the belly clipped through the floor
    return (distances < minimum_distance) | missed

def joint_pos_target_l2(env: ManagerBasedRLEnv, target: float, asset_cfg: SceneEntityCfg) -> torch.Tensor:
    """Penalize joint position deviation from a target value."""
    # extract the used quantities (to enable type-hinting)
    asset: Articulation = env.scene[asset_cfg.name]
    # wrap the joint positions to (-pi, pi)
    joint_pos = wrap_to_pi(asset.data.joint_pos[:, asset_cfg.joint_ids])
    # compute the reward
    return torch.sum(torch.square(joint_pos - target), dim=1)


def joint_effort_above_continuous_limit(
    env: ManagerBasedRLEnv, limit: float, asset_cfg: SceneEntityCfg
) -> torch.Tensor:
    """Penalize joint efforts that exceed a specified continuous limit."""
    asset = env.scene[asset_cfg.name]
    
    # Change 'applied_effort' to 'applied_torque'
    efforts = torch.abs(asset.data.applied_torque)
    
    excess = torch.clamp(efforts - limit, min=0.0)
    return torch.sum(torch.square(excess), dim=-1)

import torch
import torch.nn as nn
import torch.nn.functional as F
from rsl_rl.algorithms import Distillation
from rsl_rl.runners import OnPolicyRunner


class DistillationRolloutBuffer:
    """Lightweight rollout buffer designed specifically for grouped observation dictionaries."""

    def __init__(self, num_envs: int, num_steps: int, obs_example, device: str = "cpu"):
        self.num_envs = num_envs
        self.num_steps = num_steps
        self.device = device
        self.step = 0

        # Preallocate storage for each observation group
        self.observations = {
            group: torch.zeros(
                (num_steps, num_envs, tensor.shape[-1]),
                dtype=torch.float32,
                device=device,
            )
            for group, tensor in obs_example.items()
        }
        self.dones = torch.zeros((num_steps, num_envs), dtype=torch.bool, device=device)

    def add(self, obs, dones):
        for group, tensor in obs.items():
            self.observations[group][self.step].copy_(tensor)
        self.dones[self.step].copy_(dones)
        self.step += 1

    def clear(self):
        self.step = 0

    def get_step(self, step_idx: int):
        obs_slice = {
            group: self.observations[group][step_idx]
            for group in self.observations
        }
        return obs_slice, self.dones[step_idx]


class NativeBeliefDistillation(Distillation):
    """Custom belief distillation algorithm for training the student policy using a frozen teacher."""

    def __init__(
        self,
        student_policy,
        teacher_policy,
        lambda_bc=1.0,
        lambda_rec=0.5,
        prop_dim=45,
        **kwargs,
    ):
        super().__init__(student_policy, **kwargs)
        self.student = self.policy
        self.teacher = teacher_policy
        self.prop_dim = prop_dim

        # Attributes expected by OnPolicyRunner
        self.rnd = False
        self.empirical_normalization = False

        # Freeze teacher weights and set to evaluation mode
        self.teacher.eval()
        self.teacher.requires_grad_(False)

        self.lambda_bc = lambda_bc
        self.lambda_rec = lambda_rec
        self.custom_storage = None
        self.last_obs = None

    def init_storage(self, num_envs, num_steps, obs, *args, **kwargs):
        """Initialize custom rollout buffer for grouped observations."""
        self.custom_storage = DistillationRolloutBuffer(
            num_envs=num_envs,
            num_steps=num_steps,
            obs_example=obs,
            device=self.device,
        )

    def act(self, obs):
        """Query student actions and generate teacher target actions during rollouts."""
        self.last_obs = obs

        # 1. Query student actions
        actions = self.student.act(obs)
        self.transition.actions = actions.detach()

        # 2. Query teacher actions using privileged data
        with torch.no_grad():
            teacher_actions = self.teacher.act_inference(obs)

        self.transition.privileged_actions = teacher_actions.detach()
        return self.transition.actions

    def process_env_step(self, obs, rewards, dones, extras):
        """Record rollout step into custom storage and reset terminated environments."""
        if self.custom_storage is not None and self.last_obs is not None:
            self.custom_storage.add(self.last_obs, dones)

        # Reset student recurrent hidden states for environments that have terminated
        if hasattr(self.student, "reset"):
            self.student.reset(dones)

    def compute_returns(self, last_critic_obs):
        """Bypass advantage estimation. Distillation does not utilize value functions or returns."""
        pass

    def update(self):
        """Execute distillation learning epochs combining behavior cloning and state reconstruction losses."""
        totals = {"behavior": 0.0, "reconstruction": 0.0}
        count = 0

        for _ in range(self.num_learning_epochs):
            if hasattr(self.student, "reset"):
                self.student.reset()

            # Iterate sequentially over collected rollout transitions to preserve GRU continuity
            for step_idx in range(self.custom_storage.step):
                obs, dones = self.custom_storage.get_step(step_idx)
                privileged_targets = obs["privileged"]

                # 1. Generate teacher actions without gradients
                with torch.no_grad():
                    teacher_actions = self.teacher.act_inference(obs)

                # 2. Forward pass student model
                student_actions = self.student.act(obs)
                reconstruction = self.student.reconstruction

                # 3. Compute losses (Behavior Cloning + Privileged Reconstruction)
                behavior_loss = F.mse_loss(student_actions, teacher_actions)
                reconstruction_loss = F.mse_loss(reconstruction, privileged_targets)
                loss = (self.lambda_bc * behavior_loss) + (self.lambda_rec * reconstruction_loss)

                # 4. Backpropagation & optimization step
                self.optimizer.zero_grad()
                loss.backward()
                if self.max_grad_norm > 0:
                    nn.utils.clip_grad_norm_(self.student.parameters(), self.max_grad_norm)
                self.optimizer.step()

                # Detach recurrent state to truncate BPTT to 1 step
                if hasattr(self.student, "detach_hidden_states"):
                    self.student.detach_hidden_states()

                # Reset recurrent state for environments terminated in this step
                if hasattr(self.student, "reset"):
                    self.student.reset(dones)

                totals["behavior"] += behavior_loss.item()
                totals["reconstruction"] += reconstruction_loss.item()
                count += 1

        self.custom_storage.clear()

        mean_behavior = totals["behavior"] / max(1, count)
        mean_reconstruction = totals["reconstruction"] / max(1, count)

        # Keys mapped to satisfy default OnPolicyRunner.log formatting and TensorBoard summaries
        return {
            "surrogate": mean_behavior,
            "value_loss": mean_reconstruction,
            "behavior_loss": mean_behavior,
            "reconstruction_loss": mean_reconstruction,
        }


class NativeDistillationRunner(OnPolicyRunner):
    """Custom runner that accepts a pre-loaded teacher and initializes the native student architecture."""

    def __init__(self, env, train_cfg, teacher_policy, log_dir=None, device="cpu"):
        self.teacher_policy = teacher_policy
        # The base init automatically calls self._construct_algorithm()
        super().__init__(env, train_cfg, log_dir, device)

    def _construct_algorithm(self, obs):
        """Construct the decoupled student policy and native distillation algorithm."""
        from .belief_actor import StudentActorCritic

        # 1. Initialize Student Policy
        student_cfg = self.policy_cfg.copy()
        student_cfg.pop("class_name", None)
        self.student = StudentActorCritic(
            num_actions=self.env.num_actions,
            **student_cfg,
        ).to(self.device)

        # 2. Initialize Algorithm
        alg_cfg = self.alg_cfg.copy()
        alg_cfg.pop("class_name", None)
        alg = NativeBeliefDistillation(
            student_policy=self.student,
            teacher_policy=self.teacher_policy,
            device=self.device,
            **alg_cfg,
        )

        # 3. Initialize Custom Rollout Storage
        alg.init_storage(
            num_envs=self.env.num_envs,
            num_steps=self.num_steps_per_env,
            obs=obs,
        )
        return alg

import torch
import torch.nn as nn


class StudentActorCritic(nn.Module):
    is_recurrent = True

    def __init__(
        self,
        num_actions: int,
        prop_dim: int = 45,
        extero_dim: int = 121,
        priv_dim: int = 124,
        latent_dim: int = 64,
        rnn_hidden_dim: int = 256,
        rnn_num_layers: int = 2,
        student_hidden_dims=(512, 256, 128),
        activation: str = "elu",
        **kwargs,
    ):
        super().__init__()

        self.prop_dim = prop_dim
        self.extero_dim = extero_dim
        self.priv_dim = priv_dim
        self.rnn_hidden_dim = rnn_hidden_dim
        self.rnn_num_layers = rnn_num_layers
        self.hidden = None
        self.reconstruction = None

        act_cls = nn.ELU if activation.lower() == "elu" else nn.ReLU

        # 1. Proprioceptive encoder
        self.proprio_net = nn.Sequential(
            nn.Linear(prop_dim, 256),
            act_cls(),
            nn.Linear(256, 128),
            act_cls(),
        )

        # 2. Exteroceptive encoder
        self.extero_net = nn.Sequential(
            nn.Linear(extero_dim, 256),
            act_cls(),
            nn.Linear(256, 128),
            act_cls(),
        )

        # 3. Recurrent belief GRU
        self.rnn = nn.GRU(
            input_size=256,
            hidden_size=rnn_hidden_dim,
            num_layers=rnn_num_layers,
            batch_first=True,
        )

        # Action and reconstruction heads
        self.action_head = nn.Sequential(
            nn.Linear(rnn_hidden_dim, 128),
            act_cls(),
            nn.Linear(128, num_actions),
        )
        self.reconstruction_head = nn.Sequential(
            nn.Linear(rnn_hidden_dim, 128),
            act_cls(),
            nn.Linear(128, priv_dim),
        )

        self.register_buffer("action_std", torch.zeros(num_actions))

    def _extract_inputs(self, obs):
        if isinstance(obs, dict) or hasattr(obs, "keys"):
            proprio = obs["proprioception"]
            extero = obs["exteroception"]
        else:
            proprio = obs[:, :self.prop_dim]
            extero = obs[:, self.prop_dim:self.prop_dim + self.extero_dim]
        return proprio, extero

    def act(self, obs, **kwargs):
        proprio, extero = self._extract_inputs(obs)

        proprio_feat = self.proprio_net(proprio)
        extero_feat = self.extero_net(extero)
        fused = torch.cat([proprio_feat, extero_feat], dim=-1).unsqueeze(1)

        gru_out, self.hidden = self.rnn(fused, self.hidden)
        belief_state = gru_out.squeeze(1)

        actions = self.action_head(belief_state)
        self.reconstruction = self.reconstruction_head(belief_state)
        return actions

    def act_inference(self, obs_dict, **kwargs):
        if not hasattr(self, "_logged_inference"):
            self._logged_inference = True
            print("=" * 60)
            print(f"[INFERENCE DEBUG] obs_dict type: {type(obs_dict)}")
            if hasattr(obs_dict, "keys"):
                print(f"[INFERENCE DEBUG] Available keys: {list(obs_dict.keys())}")
                for k in obs_dict.keys():
                    v = obs_dict[k]
                    print(f"  -> key '{k}': shape {v.shape if hasattr(v, 'shape') else type(v)}")
            elif isinstance(obs_dict, torch.Tensor):
                print(f"[INFERENCE DEBUG] Tensor shape: {obs_dict.shape}")

        prop, ext = self._extract_inputs(obs_dict)

        if not hasattr(self, "_logged_extracted"):
            self._logged_extracted = True
            print(f"[INFERENCE DEBUG] Extracted prop shape: {prop.shape if hasattr(prop, 'shape') else type(prop)}")
            print(f"[INFERENCE DEBUG] Extracted ext shape: {ext.shape if hasattr(ext, 'shape') else type(ext)}")
            if hasattr(prop, "shape") and hasattr(ext, "shape"):
                print(f"[INFERENCE DEBUG] prop mean: {prop.mean().item():.3f}, std: {prop.std().item():.3f}")
                print(f"[INFERENCE DEBUG] ext mean: {ext.mean().item():.3f}, std: {ext.std().item():.3f}")
            print("=" * 60)

        p_feat = self.proprio_net(prop)
        e_feat = self.extero_net(ext)
        fused = torch.cat([p_feat, e_feat], dim=-1).unsqueeze(1)

        out, self.hidden = self.rnn(fused, self.hidden)
        latent = out.squeeze(1)

        self.reconstruction = self.reconstruction_head(latent)
        return self.action_head(latent)

    def reset(self, dones=None, hidden_states=None):
        if dones is None:
            self.hidden = None if hidden_states is None else hidden_states.detach().clone()
        elif self.hidden is not None:
            keep = (~dones.reshape(1, -1, 1).bool()).to(self.hidden.dtype)
            self.hidden = self.hidden * keep

    def get_hidden_states(self):
        return None if self.hidden is None else self.hidden.detach().clone()

    def detach_hidden_states(self):
        if self.hidden is not None:
            self.hidden = self.hidden.detach()

    def evaluate(self, obs, **kwargs):
        return self.act_inference(obs, **kwargs)

def get_latest_teacher_checkpoint() -> str:
    teacher_dir = os.path.abspath(os.path.join("logs", "rsl_rl", "QuadrupedSpider"))
    if not os.path.exists(teacher_dir):
        raise FileNotFoundError(f"Teacher directory not found at: {teacher_dir}")

    run_dirs = [
        os.path.join(teacher_dir, d)
        for d in os.listdir(teacher_dir)
        if os.path.isdir(os.path.join(teacher_dir, d))
    ]
    if not run_dirs:
        raise FileNotFoundError(f"No run directories found inside: {teacher_dir}")

    latest_run_dir = max(run_dirs, key=os.path.getmtime)
    checkpoints = glob.glob(os.path.join(latest_run_dir, "**", "model_*.pt"), recursive=True)
    if not checkpoints:
        checkpoints = glob.glob(os.path.join(latest_run_dir, "*.pt"))
    if not checkpoints:
        raise FileNotFoundError(f"No .pt checkpoint files found inside latest run: {latest_run_dir}")

    def extract_iter(p):
        base = os.path.basename(p)
        num = "".join(filter(str.isdigit, base))
        return int(num) if num else os.path.getmtime(p)

    return max(checkpoints, key=extract_iter)

class RosTeleopNode(Node):
    """ROS 2 listener node for teleoperation and manual reset commands."""

    def __init__(self):
        super().__init__("isaac_teleop_node")
        self.cmd_vel = [0.0, 0.0, 0.0]
        self.reset_requested = False

        self.twist_sub = self.create_subscription(Twist, "/cmd_vel", self._cmd_vel_callback, 10)
        self.joy_sub = self.create_subscription(Joy, "/joy", self._joy_callback, 10)

    def _cmd_vel_callback(self, msg: Twist):
        self.cmd_vel = [float(msg.linear.x), float(msg.linear.y), float(msg.angular.z)]

    def _joy_callback(self, msg: Joy):
        # Button index 0 is 'A' on standard Xbox mappings
        if len(msg.buttons) > 0 and msg.buttons[0] == 1:
            self.reset_requested = True

        env_cfg.terminations = None

    # Locate checkpoint
    log_root_path = os.path.abspath(os.path.join("logs", "rsl_rl", agent_cfg.experiment_name))
    if args_cli.checkpoint:
        resume_path = retrieve_file_path(args_cli.checkpoint)
    else:
        resume_path = get_checkpoint_path(log_root_path, agent_cfg.load_run, agent_cfg.load_checkpoint)

    env_cfg.log_dir = os.path.dirname(resume_path)

    # Initialize Environment
    env = gym.make(args_cli.task, cfg=env_cfg)
    env = RslRlVecEnvWrapper(env, clip_actions=agent_cfg.clip_actions)

    print(f"[INFO]: Loading model checkpoint from: {resume_path}")
    if agent_cfg.class_name == "OnPolicyRunner":
        runner = OnPolicyRunner(env, agent_cfg.to_dict(), log_dir=None, device=agent_cfg.device)
    elif agent_cfg.class_name == "DistillationRunner":
        runner = DistillationRunner(env, agent_cfg.to_dict(), log_dir=None, device=agent_cfg.device)
    runner.load(resume_path)

    policy = runner.get_inference_policy(device=env.unwrapped.device)

    # ROS 2 node initialization
    if not rclpy.ok():
        rclpy.init()
    ros_node = RosTeleopNode()

    print("\n" + "=" * 65)
    print("[INFO] Isaac Sim ROS 2 Teleop Listener Active:")
    print("  - Listening on /cmd_vel (Twist)")
    print("  - Listening on /joy (Button 0 'A' triggers reset)")
    print("=" * 65 + "\n")

    command_term = None
    if isinstance(env.unwrapped, ManagerBasedRLEnv) and "base_velocity" in env.unwrapped.command_manager.active_terms:
        command_term = env.unwrapped.command_manager.get_term("base_velocity")

    dt = env.unwrapped.step_dt
    obs = env.get_observations()
    loop_step = 0

    while simulation_app.is_running():
        start_time = time.time()

        # Spin ROS events non-blockingly
        rclpy.spin_once(ros_node, timeout_sec=0.0)

        with torch.inference_mode():
            # Manual reset triggered by Xbox button A
            if ros_node.reset_requested:
                print("[TELEOP] Button A pressed -> Resetting robot!")
                ros_node.reset_requested = False
                all_env_ids = torch.arange(env.unwrapped.num_envs, device=env.unwrapped.device)
                env.unwrapped.reset(env_ids=all_env_ids)
                obs = env.get_observations()

            # Push velocity command into Isaac Lab manager buffer
            cmd_tensor = torch.tensor(ros_node.cmd_vel, dtype=torch.float32, device=env.unwrapped.device)
            if command_term is not None:
                command_term.command[:] = cmd_tensor.repeat(env.unwrapped.num_envs, 1)

            # Step policy & physics
            actions = policy(obs)
            obs, _, dones, _ = env.step(actions)

            # Re-apply command tensor immediately after step
            if command_term is not None:
                command_term.command[:] = cmd_tensor.repeat(env.unwrapped.num_envs, 1)

            if hasattr(policy, "reset"):
                policy.reset(dones)

            if loop_step % 60 == 0:
                print(
                    f"[Step {loop_step}] /cmd_vel: [{ros_node.cmd_vel[0]:.2f}, {ros_node.cmd_vel[1]:.2f}, {ros_node.cmd_vel[2]:.2f}] | Action Norm: {torch.norm(actions).item():.3f}"
                )

        loop_step += 1

        sleep_time = dt - (time.time() - start_time)
        if args_cli.real_time and sleep_time > 0:
            time.sleep(sleep_time)

    ros_node.destroy_node()
    rclpy.shutdown()
    env.close()


if __name__ == "__main__":
    main()
    simulation_app.close()
